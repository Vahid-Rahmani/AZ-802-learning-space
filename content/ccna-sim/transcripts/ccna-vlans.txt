# CCNA simulator lab transcript — ccna-vlans, the router-on-a-stick lab.
# "@ <device>" selects one of the lab consoles; "> " lines are typed into it and every other line
# is the exact expected output, ending with the prompt the next command sees.
# The first pass deliberately omits VLAN 20, so the cross-VLAN ping fails, and a host console
# rejects a switch command before the same ping succeeds after the repair.
@ sw1
> enable
SW1#
> configure terminal
SW1(config)#
> hostname SW1
SW1(config)#
> vlan 10
SW1(config-vlan)#
> name DEPT-A
SW1(config-vlan)#
> exit
SW1(config)#
> interface fa0/1
SW1(config-if)#
> switchport mode access
SW1(config-if)#
> switchport access vlan 10
SW1(config-if)#
> exit
SW1(config)#
> interface fa0/2
SW1(config-if)#
> switchport mode access
SW1(config-if)#
> switchport access vlan 20
SW1(config-if)#
> exit
SW1(config)#
> interface fa0/24
SW1(config-if)#
> switchport mode trunk
SW1(config-if)#
> switchport trunk allowed vlan 10,20
SW1(config-if)#
> exit
SW1(config)#
> end
